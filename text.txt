from flask import Flask, request, redirect, session, render_template
import sqlite3
from werkzeug.security import generate_password_hash, check_password_hash
import os

app = Flask(__name__)
app.secret_key = "change-this-secret-key"

DATABASE = "infrascan.db"


def get_db():
    conn = sqlite3.connect(DATABASE)
    conn.row_factory = sqlite3.Row
    return conn


def init_database():

    conn = get_db()

    # Users table
    conn.execute("""
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT UNIQUE NOT NULL,
            password TEXT NOT NULL,
            name TEXT NOT NULL,
            role TEXT NOT NULL,
            class_name TEXT
        )
    """)

    # Complaints table
    conn.execute("""
        CREATE TABLE IF NOT EXISTS complaints (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER NOT NULL,
            classroom TEXT NOT NULL,
            issue_type TEXT NOT NULL,
            description TEXT NOT NULL,
            photo TEXT,
            status TEXT NOT NULL DEFAULT 'Pending',
            admin_note TEXT,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY (user_id) REFERENCES users(id)
        )
    """)

    # Create initial Admin account
    admin = conn.execute(
        "SELECT id FROM users WHERE username = ?",
        ("admin",)
    ).fetchone()

    if admin is None:
        conn.execute("""
            INSERT INTO users
            (username, password, name, role, class_name)
            VALUES (?, ?, ?, ?, ?)
        """, (
            "admin",
            generate_password_hash("InfraScan@123"),
            "InfraScan Admin",
            "admin",
            None
        ))

    # Create initial CR test account
    cr = conn.execute(
        "SELECT id FROM users WHERE username = ?",
        ("CSE1A001",)
    ).fetchone()

    if cr is None:
        conn.execute("""
            INSERT INTO users
            (username, password, name, role, class_name)
            VALUES (?, ?, ?, ?, ?)
        """, (
            "CSE1A001",
            generate_password_hash("CR@12345"),
            "CSE 1A CR",
            "cr",
            "CSE - 1A"
        ))

    conn.commit()
    conn.close()


# =========================================
# HOME
# =========================================

@app.route("/")
def home():
    return render_template("index.html")


# =========================================
# STUDENT / CR LOGIN
# =========================================

@app.route("/student-login", methods=["GET", "POST"])
def student_login():

    if request.method == "POST":

        username = request.form.get("username", "").strip()
        password = request.form.get("password", "")

        conn = get_db()

        user = conn.execute(
            "SELECT * FROM users WHERE username = ? AND role = ?",
            (username, "cr")
        ).fetchone()

        conn.close()

        if user and check_password_hash(user["password"], password):

            session["user_id"] = user["id"]
            session["username"] = user["username"]
            session["name"] = user["name"]
            session["role"] = user["role"]

            return redirect("/student-dashboard")

        return render_template(
            "student-login.html",
            error="Invalid Student ID or password."
        )

    return render_template("student-login.html")


# =========================================
# STUDENT / CR DASHBOARD
# =========================================

@app.route("/student-dashboard")
def student_dashboard():

    if "user_id" not in session or session.get("role") != "cr":
        return redirect("/student-login")

    return render_template(
        "student-dashboard.html",
        name=session.get("name"),
        class_name="CSE - 1A"
    )




# =========================================
# NEW COMPLAINT
# =========================================

@app.route("/new-complaint", methods=["GET", "POST"])
def new_complaint():

    if "user_id" not in session or session.get("role") != "cr":
        return redirect("/student-login")

    if request.method == "POST":

        classroom = request.form.get("classroom", "").strip()
        issue_type = request.form.get("issue_type", "").strip()
        description = request.form.get("description", "").strip()

        photo = request.files.get("photo")
        photo_name = None

        # Save uploaded photo
        if photo and photo.filename:
            os.makedirs("uploads", exist_ok=True)

            photo_name = photo.filename
            photo.save(os.path.join("uploads", photo_name))

        conn = get_db()

        conn.execute("""
            INSERT INTO complaints
            (user_id, classroom, issue_type, description, photo, status)
            VALUES (?, ?, ?, ?, ?, ?)
        """, (
            session["user_id"],
            classroom,
            issue_type,
            description,
            photo_name,
            "Pending"
        ))

        conn.commit()
        conn.close()

        return redirect("/student-dashboard")

    return render_template("new-complaint.html")
